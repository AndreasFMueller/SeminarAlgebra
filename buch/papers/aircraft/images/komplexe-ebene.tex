%         Kreuz Dutch Roll)

\documentclass[tikz]{standalone}
\usepackage{amsmath}
\usepackage{times}
\usepackage{txfonts}
\usepackage{pgfplots}
\usepackage{csvsimple}
\usetikzlibrary{arrows,intersections,math,calc,plotmarks}
\definecolor{darkred}{rgb}{0.8,0,0}
\definecolor{darkgreen}{rgb}{0,0.6,0}
\begin{document}
    \def\skala{1}
    \def\sx{0.75}   % cm pro Einheit Realteil
    \def\sy{0.75}   % cm pro Einheit Imaginaerteil
    \begin{tikzpicture}[>=latex,thick,scale=\skala,
        x=\sx cm, y=\sy cm]

% Gitter
        \draw[gray!30,thin] (-9,-3) grid[xstep=1,ystep=1] (1,3);

% Achsen
        \draw[->] (-9.3,0) -- (1.3,0) node[right] {$\sigma$};
        \draw[->] (0,-3.3) -- (0,3.3) node[above] {$\omega$};
        \foreach \x in {-8,-6,-4,-2} {
            \draw (\x,-0.1) -- (\x,0.1);
            \node[below] at (\x,-0.1) {$\x$};
        }
        \foreach \y in {-3,-2,-1,1,2,3} {
            \draw (-0.1,\y) -- (0.1,\y);
            \node[left] at (-0.1,\y) {$\y$};
        }

% Propellerflugzeug (Nelson), darkred
        \draw[darkred] plot[mark=*,mark size=2.2pt]        coordinates {(-8.435,0)};
        \draw[darkred] plot[mark=square*,mark size=2pt]    coordinates {(-0.00877,0)};
        \draw[darkred] plot[only marks,mark=x,mark size=3.5pt] coordinates {(-0.487,2.335) (-0.487,-2.335)};

% Business Jet (Stengel), blue
        \draw[blue] plot[mark=*,mark size=2.2pt]           coordinates {(-1.2,0)};
        \draw[blue] plot[mark=square*,mark size=2pt]       coordinates {(0.00883,0)};
        \draw[blue] plot[only marks,mark=x,mark size=3.5pt] coordinates {(-0.116,1.39) (-0.116,-1.39)};

% Legende rechts neben der Grafik: Farbe = Flugzeug, Marker = Bewegungsform
        \begin{scope}[shift={(2.2,2.6)}]
        \draw[darkred] plot[mark=*,mark size=2.2pt] coordinates {(0,0)};
        \node[right] at (0.3,0) {\small Propellerflugzeug};
        \draw[blue] plot[mark=*,mark size=2.2pt] coordinates {(0,-0.6)};
        \node[right] at (0.3,-0.6) {\small Business Jet};
        \end{scope}
        \begin{scope}[shift={(2.2,0.6)}]
        \draw plot[mark=*,mark size=2.2pt] coordinates {(0,0)};
        \node[right] at (0.3,0) {\small Rollbewegung};
        \draw plot[mark=square*,mark size=2pt] coordinates {(0,-0.6)};
        \node[right] at (0.3,-0.6) {\small Spiralbewegung};
        \draw plot[mark=x,mark size=3.5pt] coordinates {(0,-1.2)};
        \node[right] at (0.3,-1.2) {\small Dutch Roll};
        \end{scope}

    \end{tikzpicture}
\end{document}
