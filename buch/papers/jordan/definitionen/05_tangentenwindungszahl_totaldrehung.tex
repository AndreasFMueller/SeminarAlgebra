\begin{definition}[Tangentenwindungszahl / Totaldrehung\label{jordan:def:tangentenwindungszahl_totaldrehung}]
    Für eine reguläre C\textsuperscript{1}-Kurve $\gamma$ ist die Totaldrehung, auch bekannt als Tangentenwindungszahl, der Grad der Tangentenabbildung $\tau: S\textsuperscript{1} \to S\textsuperscript{1}$:

    \[
        \deg(\tau) = \frac{1}{2\pi} \oint  \mathcal{K}(s) ds
    \]
    wobei $\mathcal{K}(s)$ die Krümmung der Kurve $\gamma$ bezüglich der Bogenlängenparametrisierung ist.
\end{definition}
