\begin{definition}[Reell-analytische Kurve\label{jordan:definition:reell_analytische_kurve}]
\index{Kurve!reell-analytisch}%
\index{reell-analytische Kurve}%
    Eine Kurve $\gamma\colon I \to \mathbb{R}^n$, wobei $I$ ein offenes Intervall in $\mathbb{R}$ ist, heisst \emph{reell-analytisch}, wenn für jeden Punkt $t_{0} \in I$ eine Umgebung $U \subset I$ existiert, sodass die Kurve in dieser Umgebung durch eine konvergente Potenzreihe dargestellt werden kann.
\end{definition}
