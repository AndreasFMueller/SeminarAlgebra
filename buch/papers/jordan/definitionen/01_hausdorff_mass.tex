\begin{definition}[Hausdorff-Mass\label{jordan:definition:hausdorff_mass}]
    Das $s$-dimensionale Hausdorff-Mass $\mathcal{H}^s(A)$ einer Teilmenge $A$ eines metrischen Raumes ist definiert als
    \[
        \mathcal{H}^s(A) = \lim_{\delta \to 0} \inf \left\{ \sum_{i} (\text{diam}(U_i))^s : A \subseteq \bigcup_{i} U_i, \text{diam}(U_i) < \delta \right\},
    \]
    wobei die Infimum über alle Überdeckungen von $A$ durch Mengen $U_i$ mit Durchmesser kleiner als $\delta$ genommen wird.
    
\end{definition}
