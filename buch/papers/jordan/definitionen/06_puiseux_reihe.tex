 \begin{definition}[Puiseux-Reihe\label{jordan:definition:puiseux_reihe}]
    Eine \emph{Puiseux-Reihe} ist eine Potenzreihe der Form
\index{Puiseux-Reihe}%
    \[
        f(z) = \sum_{k=k_0}^{\infty} a_k z^{\frac{k}{n}},
    \]
    wobei $n \in \mathbb{N}$, $k_0 \in \mathbb{Z}$ und $a_k \in \mathbb{C}$ für alle $k$ gilt.
 
 \end{definition}
