    \coordinate (C) at (1,3);

    \draw[dashed] (B) -- (D);

    \node[below] at (A) {$A$};
    \node[above right] at (B) {$B$};
    \node[above] at (C) {$C$};
    \node[above left] at (D) {$D$};
    \node[below left] at (E) {$E$};
